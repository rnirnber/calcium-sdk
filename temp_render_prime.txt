		root_idx := root_idx + 1;

				END;
				IF ((enter_else) AND (stop_work2 <> TRUE)) THEN					
					all_pages[0] := DUPE_LIST(these_lines);
					these_lines := DUPE_LIST({});
                    y_offset := 10;
                    x_offset := 10;
					LOCAL curr_cnt := GET_LIST_SIZE(current_words);
                    subtract_num := (curr_cnt + 1);
                    current_word := {};
                    current_words := {};					
					
				END;
			END;
			IF d == -1 THEN
				LOCAL stop_early3 := FALSE;
				IF (y_offset + line_height <= max_y) THEN					
					these_lines[0] := DUPE_LIST(current_words);
					current_words := {};
					y_offset := y_offset + line_height;
					x_offset := 10;
					stop_early3 := TRUE;
				END;
				IF stop_early3 <> TRUE THEN
					all_pages[0] := DUPE_LIST(these_lines);
				END;
			END;
		END;
	END;	
END;
