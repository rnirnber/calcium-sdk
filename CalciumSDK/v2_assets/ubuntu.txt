    [STAThread]
    static void Main(string[] args)
    {
        var app = Gtk.Application.New("org.gir.core.[__PROJECT_NAME]", ApplicationFlags.FlagsNone);
        _App = app;
        
        app.OnActivate += (sender, e) =>
        {
            _WinWidth = GetWinWidth();
            _WinHeight = GetWinHeight();
            _XScaleFactor =  GetXScaleFactor(_WinWidth);
            _YScaleFactor = GetYScaleFactor(_WinHeight);
            
            var win = Gtk.ApplicationWindow.New((Gtk.Application)sender!);
            _Window = win;
            _Window.Fullscreen();
            win.Title = "[__PROJECT_NAME]";

            var fixed_area = new Fixed();
            fixed_area.Put(AppCanvas, 0, 0);
            win.Child = fixed_area;
            
            var keyController = EventControllerKey.New();

            keyController.OnKeyReleased += ((s, e) =>
            {
                var key = e.Keyval;
                if (_AppMode == 3)
                {
                    if (key == 65363 && _AppMode == 3)
                    {
                        _CurrentDialog.Next();
                    }

                    if (key == 65361 && _AppMode == 3)
                    {
                        _CurrentDialog.Prev();
                    }
                    AppCanvas.QueueDraw();
                }
                if (_AppMode == 4 && key == 65362)
                {
                    if (CurrentScreenY > 0)
                    {
                        CurrentScreenY--;
                    
                        CurrentPlayerY--;
                        CurrentPlayerDir = 1;
                        AppCanvas.QueueDraw();   
                    }
                    else
                    {
                        CurrentPlayerY--;
                        CurrentPlayerDir = 1;
                        AppCanvas.QueueDraw();
                    }
                }
                if (_AppMode == 4 && key == 65364)
                {
                    if (CurrentScreenY < 3)
                    {
                        CurrentScreenY++;
                        CurrentPlayerY++;
                        CurrentPlayerDir = 2;
                    }
                    else
                    {
                        CurrentPlayerY++;
                        CurrentPlayerDir = 2;
                        
                    }
                    AppCanvas.QueueDraw();
                    
                }
                if (_AppMode == 4 && key == 65361)
                {
                    if (CurrentScreenX > 0)
                    {
                        CurrentScreenX--;
                        CurrentPlayerX--;
                        CurrentPlayerDir = 3;   
                    }
                    else
                    {
                        CurrentPlayerX--;
                        CurrentPlayerDir = 3;   
                    }
                    AppCanvas.QueueDraw();
                }

                if (_AppMode == 4 && key == 65363)
                {
                    if (CurrentScreenX < 5)
                    {
                        CurrentPlayerDir = 4;
                        CurrentScreenX++;
                        CurrentPlayerX++;   
                    }
                    else
                    {
                        CurrentPlayerDir = 4;
                        CurrentPlayerX++;
                    }
                    AppCanvas.QueueDraw();
                }
             });
        
            _Window.AddController(keyController);
            
            win.Present();
            
            AppCanvas.SetDrawFunc((area, ctx, width, height) =>
            {
                ctx.Antialias = Antialias.Best;
                switch (_AppMode)
                {
                    case 1:
                        if (_LastAppMode != _AppMode)
                        {
                            _AppMode = 1;
                            _LastAppMode = 1;
                            RenderMainMenu(ctx);
                        }
                        break;

                    case 2:
                        if (_CurrentDialog == null)
                        {
                            _CurrentDialog = MakeDialog(GetInitialDialog(_CurrentScene), (() =>
                            {
                                _AppMode = 1;
                                AppCanvas.QueueDraw();
                            }));
                            _CurrentDialog.Render(ctx);                            
                            _AppMode = 3;
                        }
                        break;
                    case 3:
                        _CurrentDialog.Render(ctx);
                        break;
                    case 4:
                        if (CurrentPlayerX == -1)
                        {
                            EnterScene1();
                        }
                        RenderMap(ctx);
                        RenderPlayer(ctx);
                        break;
                    default:
                        break;
                }
            });
        };
        app.RunWithSynchronizationContext(args);

    }

    private static double _GetRedAsDecimal(int red)
    {
        return ((double) red / (double) 255);
    }
    private static double _GetGreenAsDecimal(int green)
    {
        return ((double) green / (double) 255);
    }
    private static double _GetBlueAsDecimal(int blue)
    {
        return ((double) blue / (double) 255);
    }

    private static void RenderPlayer(Context ctx)
    {
        if (CurrentPlayerDir == 1)
        {
            Renderplayerup(ctx, CurrentScreenX, CurrentScreenY);
            return;
        }

        if (CurrentPlayerDir == 2)
        {
            Renderplayerdown(ctx, CurrentScreenX, CurrentScreenY);
            return;
        }
        if (CurrentPlayerDir == 3)
        {
            Renderplayerleft(ctx, CurrentScreenX, CurrentScreenY);
            return;
        }
        if (CurrentPlayerDir == 4)
        {
            Renderplayerright(ctx, CurrentScreenX, CurrentScreenY);
            return;
        }
    }
    private static void RenderMap(Context ctx)
    {
        var x_off = 0;
        var y_off = 0;
        var map_x = CurrentPlayerX - CurrentScreenX;
        var map_y = CurrentPlayerY - CurrentScreenY;
        
        // ISSUE IS THAT MAP_X needs to change from 16 + 17 + 18
        for (int i = 0; i < 6; i++)
        {
            for (int k = 0; k < 4; k++)
            {
                var tile_idx = CurrentSceneTiles.FindIndex((t) => t == ((map_x + i) * 1000 * 10) + map_y + k);
                var tile_num = -1;
                if (tile_idx == -1)
                {
                    tile_num = GetDefaultTileForScene(_CurrentScene);
                }
                else
                {
                    tile_num = CurrentSceneTiles[tile_idx + 1];
                }
                RenderAsset(tile_num, i, k, ctx, 255, 255, 255);
                y_off++;
            }
            x_off++;
        }
    }

    static void RenderMainMenu(Context ctx)
    {
        int[] rects = [__MAIN_MENU_RECTS];

        var white_alpha_red = _GetRedAsDecimal([__TRUE_ALPHA_RED]);
        var white_alpha_green = _GetGreenAsDecimal([__TRUE_ALPHA_GREEN]);
        var white_alpha_blue = _GetBlueAsDecimal([__TRUE_ALPHA_BLUE]);

        var black_alpha_red = _GetRedAsDecimal(0);
        var black_alpha_green = _GetGreenAsDecimal(0);
        var black_alpha_blue = _GetBlueAsDecimal(0);

        var use_black_background = [__USE_MAIN_MENU_BLACK_BACKGROUND];
        var stop_at = [__MAIN_MENU_STOP_AT];

        if (use_black_background)
        {
            ctx.SetSourceRgb(black_alpha_red, black_alpha_green, black_alpha_blue);
        }
        else
        {
            ctx.SetSourceRgb(white_alpha_red, white_alpha_green, white_alpha_blue);
        }
        ctx.Rectangle(0, 0, 318 * _XScaleFactor, 212 * _YScaleFactor);
        ctx.Fill();
        
        if (use_black_background)
        {
            ctx.SetSourceRgb(white_alpha_red, white_alpha_green, white_alpha_blue);
        }
        else
        {
            ctx.SetSourceRgb(black_alpha_red, black_alpha_green, black_alpha_blue);
        }
        for (int i = 0; i <= stop_at; i += 3)
        {
            var y_start = rects[i] * _YScaleFactor;
            var x_start = rects[i + 1] * _XScaleFactor;
            var x_end = rects[i + 2] * _XScaleFactor;
            ctx.Rectangle(x_start, y_start, ((x_end - x_start) + _XScaleFactor), _YScaleFactor);
            ctx.Fill();
        }
    }
    public static int GetWinWidth()
    {
        var display = Gdk.Display.GetDefault();
        if (display != null)
        {
            var monitorsModel = display.GetMonitors();
            if (monitorsModel.GetNItems() > 0)
            {
                nint monitorPtr = monitorsModel.GetItem(0); 
                var monitor = ((GObject.Internal.InstanceWrapper.WrapHandle<Gdk.Monitor>(monitorPtr, false) as Gdk.Monitor));
                
                if (monitor != null)
                {
                    
                    var current_width = 0;
                    var current_height = 0;
                    var done = false;
                    while (!done)
                    {
                        if ((current_width + 318) <= monitor.Geometry.Width && (current_height + 212) <=  monitor.Geometry.Height)
                        {
                            current_width += 318;
                            current_height += 212;
                            continue;
                        }
                        return current_width;
                    }
                }
            }
        }

        return 318;
    }

    public class TextDialog()
    {
        public int CurrentPage = 0;
        public System.Action OnFinished = (() => { });
        public List<TextDialogPage> Pages = new List<TextDialogPage>();
        private Context _ctx;

        public void Render(Context ctx)
        {
            var x_offset = 10;
            var y_offset = 10;
            var max_x = 318 - 10;
            var max_y = 212 - 10;
            var wordspace_size = 5;
            var letterspace_size = 2;
            var line_height = 14;
            _ctx = ctx;
            
            ctx.SetSourceRgb(0, 0, 0);
            ctx.Rectangle(0, 0, 318 * _XScaleFactor, 212 * _YScaleFactor);
            ctx.Fill();
            ctx.SetSourceRgb(1, 1, 1);
            
            
            Pages[CurrentPage].Lines.ForEach((l) =>
            {
                var chars = new List<char>();
                l.Words.ForEach((w) =>
                {
                    w.ForEach((w2) =>
                    {
                        chars.Add((char) w2);
                    });
                });
                var test = l.Words.Select((w) => w.Select((i) => ((char)i))).ToList();
                l.Words.ForEach((w) =>
                {
                    w.ForEach((c) =>
                    {
                        RenderChar(c, x_offset, y_offset, 255, 255, 255 ,ctx);
                        x_offset += GetCharacterWidth(c);
                        x_offset += letterspace_size;     
                    });
                    x_offset += wordspace_size;
                });
                x_offset = 10;
                ctx.Fill();
                y_offset += line_height;
            });
        }

        public void Next()
        {
            if (this.CurrentPage == this.Pages.Count - 1)
            {
                this.OnFinished();
                return;
            }

            this.CurrentPage++;
            Render(_ctx);
        }

        public void Prev()
        {
            if (this.CurrentPage > 0)
            {
                this.CurrentPage--;
                Render(_ctx);
            }
        }
    }

    public static TextDialog MakeDialog(List<int> data, System.Action on_finished = null)
    {
        if (on_finished == null)
        {
            on_finished = (() => { });
        }
        var ret = new TextDialog();
        ret.OnFinished = on_finished;
        var page_cnt = 1;
        var x_offset = 10;
        var y_offset = 10;

        var max_x = 318 - 10;
        var max_y = 212 - 10;

        var wordspace_size = 5;
        var letterspace_size = 2;
        var line_height = 14;

        var current_page = new TextDialogPage();
        var current_line = new TextDialogLine();
        var current_words = new List<List<int>>();
        var current_word = new List<int>();
        
        current_page.Lines = new List<TextDialogLine>();

        for (int i = 0; i < data.Count; i++)
        {
            var d = data[i];
            
            if (d == 32)
            {
                current_words.Add(current_word.ToList());
                current_word = new List<int>();
                x_offset += wordspace_size;
                continue;
            }
            if (d != -1)
            {
                var this_width = GetCharacterWidth(d);
                var this_letterspace_size = letterspace_size + 0;
                var new_x = x_offset + this_width;
                if (new_x <= max_x)
                {
                    current_word.Add(d);
                    x_offset += this_width + this_letterspace_size;
                }
                else
                {
                    if (y_offset + line_height <= max_y)
                    {
                        var d_line = new TextDialogLine();
                        d_line.Words = current_words.ToList();
                        var curr_cnt = current_word.Count;
                        current_word = new List<int>();
                        current_words = new List<List<int>>();
                        current_page.Lines.Add(TextDialogLine.CreateDupe(d_line));
                        y_offset += line_height;
                        x_offset = 10;
                        i -= curr_cnt;
                        i--;
                        continue;
                    }
                    else
                    {
                        var d_line = new TextDialogLine();
                        d_line.Words = current_words.ToList();
                        current_page.Lines.Add(TextDialogLine.CreateDupe(d_line));
                        ret.Pages.Add(TextDialogPage.MakeDupe(current_page));
                        y_offset = 10;
                        x_offset = 10;
                        i -= current_word.Count;
                        i--;
                        current_word = new List<int>();
                        current_words = new List<List<int>>();
                        current_page = new TextDialogPage();
                    }
                }
            }
            else
            {
                if (y_offset + line_height <= max_y)
                {
                    var d_line = new TextDialogLine();
                    current_words.Add(current_word);
                    d_line.Words = current_words.ToList();
                    var curr_cnt = current_word.Count;
                    current_word = new List<int>();
                    current_words = new List<List<int>>();
                    current_page.Lines.Add(TextDialogLine.CreateDupe(d_line));
                    y_offset += line_height;
                    x_offset = 10;
                    
                }
                else
                {
                    ret.Pages.Add(TextDialogPage.MakeDupe(current_page));
                    x_offset = 10;
                    y_offset = 10;
                    current_word = new List<int>();
                    current_words = new List<List<int>>();
                    current_page = new TextDialogPage();
                }
            }
            if (i == data.Count - 1)
            {
                var d_line = new TextDialogLine();
                d_line.Words = current_words.ToList();
                var curr_cnt = current_word.Count;
                current_word = new List<int>();
                current_words = new List<List<int>>();
                current_page.Lines.Add(TextDialogLine.CreateDupe(d_line));
                
                ret.Pages.Add(TextDialogPage.MakeDupe(current_page));
            }
            
            
        }
        
        return ret;
    }
